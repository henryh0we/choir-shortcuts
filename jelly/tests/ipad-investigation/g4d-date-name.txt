// Installed name: TEST - G4D Date Name
// Purpose/role: Generate date formatting and editable name default; independent generation diagnostic.
// Specification revision: 1; interface 1.0. Contract: Choose specified test Date; inspect formatted result; add suffix - TEST to editable default; preview edited trimmed name; no writes/explicit return
// Dependencies: None. Exact installed targets must match source; export filenames do not set installed names.
// Side effects: Previews/prompts only; no saves, renames, imports, metadata or setlist writes.
// Target: iPadOS/Jellycuts versions must be recorded; previous profile iPadOS 26.6 / Jellycuts 2026.6.4. Local comparison Open-Jellycore e1ec31c / Swift 6.4.1.
// Stage: Candidate, not iPad compiled/imported/executed. Local build results: generation-groups-builds.md. Test instructions/results: generation-groups.md.
// Setup: No native fixture slots. Import unchanged first; do not repair before recording the untouched result.
import Shortcuts

askForInput(prompt: "G4D choose test date from the checklist: 31 Dec 2026, then rerun for 1 Jan 2027.", type: Date, default: "", allowDecimal: false, allowNegative: false) >> serviceDate
formatDate(date: "${serviceDate}", dStyle: Custom, tStyle: None, custom: "yyyy-MM-dd - EEE", isoTime: false) >> formattedDate
quicklook(input: formattedDate)
var selectedServiceType = "Evensong"
text(text: "${formattedDate} - ${selectedServiceType}") >> proposedName
quicklook(input: proposedName)
askForInput(prompt: "G4D inspect default, then append - TEST to confirm edited value is used.", type: Text, default: "${proposedName}", allowDecimal: false, allowNegative: false) >> enteredName
replaceText(input: "${enteredName}", find: "^[ ]+|[ ]+$", replace: "", isRegex: true, caseSensitive: true) >> trimmedName
quicklook(input: trimmedName)
