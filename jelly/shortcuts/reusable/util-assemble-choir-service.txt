// Installed name: UTIL - Assemble Choir Service
// Purpose: Generate standard actions; insert forScore actions at FS01-FS13.
// Role: Reusable; specification revision 7; completed interface 1.1.
// Input: Text interactive. Other/missing input returns instructions; direct no-input unsupported.
// Output: One service-report Text if native actions/navigation complete.
// Side effects after assembly: File selection, clean archive/import, metadata, setlist writes.
// Called mode is interactive; cancellation/failure may leave partial writes with no report.
// Dependencies: UTIL - Canonical Choir Name and UTIL - Archive Choir PDF, interface 2.0.
// Sources: util-canonical-choir-name.jelly and util-archive-choir-pdf.jelly.
// Target: iPadOS 26.6, forScore 15.2.5, Jellycuts 2026.6.4; Files/app permissions.
// Stage: Drafted; local docs/source reviewed, not compiled/device tested.
// Assembly: ../../docs/config-integration.md; tests: ../../tests/config-integration.md

import Shortcuts

// RECEIVING: Set input to Text for interactive mode. Keep native File/Folder/Dictionary tokens inside archiveInput and both Text items inside sourceNameInput. Final exit returns the serviceReport token.

// ASSEMBLY GUARD: Keep the first Stop and Output below until all CONF01-CONF02, COL01-COL03, SCORE01-SCORE02 and FS01-FS13 slots are complete, native outputs reconnected, and archive CFG02-CFG04 complete. Then remove only this first Stop and Output.
text(text: "Service helper needs its forScore assembly slots completed.") >> stopMessage1
exit(var: stopMessage1)

if(ShortcutInput != "interactive") {
    text(text: "Launch MAIN - Build Choir Service.") >> stopMessage2
    exit(var: stopMessage2)
}
// CONF01 CHOIR MUSIC ROOT: replace next Nothing with Folder selecting Choir Music, NOT Library. Reconnect Get Contents of Folder(root) below and CONF02 to that native Folder output.
nothing() >> choirRoot
// CONF02 CONFIG FILE: replace next Nothing with native Get File from Folder; Folder=choirRoot, path=config.json, error if absent ON. Reconnect Get Dictionary from Input below to the actual config File. Do not select a separate unrelated config file.
nothing() >> configFile
// Invalid JSON halts here before PDF selection, saves, imports or setlist writes.
getDictionaryFrom(input: configFile) >> workflowConfig
valueFor(key: "version", dictionary: workflowConfig) >> configVersion
text(text: "${configVersion}") >> configVersionText
if(configVersionText != "1") {
    text(text: "Only config.json version 1 is supported. No changes made.") >> stopMessage3
    exit(var: stopMessage3)
}
valueFor(key: "prefixRegex", dictionary: workflowConfig) >> prefixRegex
valueFor(key: "categories", dictionary: workflowConfig) >> categories
valueFor(key: "libraryFolder", dictionary: workflowConfig) >> libraryFolderName
if(prefixRegex == nil) {
    text(text: "config.json needs a nonempty prefixRegex. No changes made.") >> stopMessage4
    exit(var: stopMessage4)
}
if(categories == nil) {
    text(text: "config.json needs a nonempty categories array. No changes made.") >> stopMessage5
    exit(var: stopMessage5)
}
if(libraryFolderName == nil) {
    text(text: "config.json needs libraryFolder. No changes made.") >> stopMessage6
    exit(var: stopMessage6)
}
// Validate regex syntax before any writes; a malformed expression halts this action.
replaceText(input: "ConfigValidation.pdf", find: "${prefixRegex}", replace: "", isRegex: true, caseSensitive: false) >> regexValidation
getFolderContents(folder: choirRoot, recursive: false) >> rootChildren
nothing() >> noLibraryFolder
var libraryRoot = noLibraryFolder
repeatEach(rootChildren) {
    fileDetail(input: RepeatItem, property: Name) >> rootChildName
    if(rootChildName == "${libraryFolderName}") {
        if(libraryRoot != nil) {
            text(text: "Multiple root children match libraryFolder. No changes made.") >> stopMessage7
            exit(var: stopMessage7)
        }
        libraryRoot = RepeatItem
    }
}
if(libraryRoot == nil) {
    text(text: "Choir Music has no folder named ${libraryFolderName}. No changes made.") >> stopMessage8
    exit(var: stopMessage8)
}
// Resolve a native Folder before allowing archive/import operations. Never concatenate a provider path.
getFolderContents(folder: libraryRoot, recursive: false) >> libraryPreflight
valueFor(key: "serviceType", dictionary: workflowConfig) >> serviceTypes
var selectedServiceType = "Other"
if(serviceTypes != nil) {
    choose(list: serviceTypes, prompt: "Choose the service type from config.json.", multiple: false, all: false) >> chosenServiceType
    if(chosenServiceType == nil) {
        text(text: "No service type selected. No changes made.") >> stopMessage9
        exit(var: stopMessage9)
    }
    selectedServiceType = chosenServiceType
}
// servicesFolder and manifestFilename are reserved; this release creates no manifest or service-folder copies.
selectFile(type: Files, multiple: true) >> selectedFiles
if(selectedFiles == nil) {
    text(text: "No files selected. No changes made.") >> stopMessage10
    exit(var: stopMessage10)
}
count(type: Items, input: selectedFiles) >> pieceCount
nothing() >> emptyOrderedFiles
var orderedFiles = emptyOrderedFiles
repeatEach(selectedFiles) {
    choose(list: selectedFiles, prompt: "Choose piece ${RepeatIndex} of ${pieceCount} in running order.", multiple: false, all: false) >> orderedFile
    // COL01 ORDERED FILES: Replace next Nothing with native Add to Variable, input=orderedFile from this Choose from List, variable=orderedFiles. Append one native File per iteration; do not use Text or Set Variable.
    nothing()
}

nothing() >> emptyOrderedNames
var orderedNames = emptyOrderedNames
repeatEach(orderedFiles) {
    fileDetail(input: RepeatItem, property: File Extension) >> sourceExtension
    changeCase(text: "${sourceExtension}", case: lowercase) >> lowerExtension
    if(lowerExtension != "pdf") {
        text(text: "A selected item is not a PDF. No changes made.") >> stopMessage11
        exit(var: stopMessage11)
    }
    fileDetail(input: RepeatItem, property: Name) >> orderedName
    // COL02 ORDERED NAMES: Replace next Nothing with native Add to Variable, input=orderedName, variable=orderedNames. Append the filename Text in this validation loop.
    nothing()
}
combineText(text: [orderedNames], combine: Custom, separator: " | ") >> runningOrder
askForInput(prompt: "Service name including date, for example 2026-10-18 - Evensong", type: Text, default: "", allowDecimal: false, allowNegative: false) >> enteredServiceName
replaceText(input: "${enteredServiceName}", find: "^[ ]+|[ ]+$", replace: "", isRegex: true, caseSensitive: true) >> serviceName
if(serviceName == nil) {
    text(text: "A service name is required. No changes made.") >> stopMessage12
    exit(var: stopMessage12)
}
alert(alert: """
${serviceName}
Service type: ${selectedServiceType}

${runningOrder}

Continue to save clean PDFs and prepare scores. Check omissions/repetitions now. Setlist entries change only after every score is ready.
""", title: "Confirm service", cancel: true)

nothing() >> emptyPreparedScores
var preparedScores = emptyPreparedScores
repeatEach(orderedFiles) {
    fileDetail(input: RepeatItem, property: Name) >> sourceName
    list(items: [sourceName, prefixRegex]) >> sourceNameInput
    runShortcut(name: "UTIL - Canonical Choir Name", input: sourceNameInput, show: false) >> canonicalStem
    if(canonicalStem == nil) {
        text(text: "Empty music identity. Earlier pieces may already be prepared.") >> stopMessage13
        exit(var: stopMessage13)
    }
    // Native context List: keep File, Folder and Dictionary as whole variable tokens; no Text/JSON conversion.
    list(items: [RepeatItem, libraryRoot, workflowConfig]) >> archiveInput
    runShortcut(name: "UTIL - Archive Choir PDF", input: archiveInput, show: false) >> cleanFile
    if(cleanFile == nil) {
        text(text: "No clean PDF returned for ${canonicalStem}. Inspect archive before retrying.") >> stopMessage14
        exit(var: stopMessage14)
    }

    // FS01 LOOKUP. Replace next Nothing with Get score named canonicalStem. Continue if not found ON; category filters unset; Reference empty. Reconnect the following existingScore presence If and SCORE01 list input to THIS Get score output. Never use the first score of the whole library.
    nothing() >> existingScore

    nothing() >> noWorkingScore
    var workingScore = noWorkingScore
    if(existingScore != nil) {
        // SCORE01 SCORE RESULT: Replace next Nothing with native Get Item from List, List=existingScore from its individual lookup, Item at Index=1. Reconnect the following Set Variable workingScore input to this native Score before deleting Nothing.
        nothing() >> reusedWorkingScore
        workingScore = reusedWorkingScore
    } else {
        // FS02 IMPORT. Replace next Nothing with the import action that worked in the device diagnostic. Files=cleanFile native File; Conflict Handling=Skip. Raw import output is not consumed.
        nothing()
        // FS03 LOOKUP AFTER IMPORT. Replace next Nothing with Get score named canonicalStem, Continue if not found ON, other filters unset. Reconnect the following importedScore presence If and SCORE02 list input to this lookup's native Score output.
        nothing() >> importedScore
        if(importedScore == nil) {
            text(text: "Import unconfirmed for ${canonicalStem}. Inspect forScore before retrying. Setlist entries have not been replaced.") >> stopMessage15
            exit(var: stopMessage15)
        }
        // SCORE02 SCORE RESULT: Replace next Nothing with native Get Item from List, List=importedScore from its individual lookup, Item at Index=1. Reconnect the following Set Variable workingScore input to this native Score before deleting Nothing.
        nothing() >> importedWorkingScore
        workingScore = importedWorkingScore
    }

    var metadataMode = "Review metadata"
    if(existingScore != nil) {
        list(items: ["Reuse unchanged", "Review metadata"]) >> metadataChoices
        choose(list: metadataChoices, prompt: "Existing score: preserve or review metadata?", multiple: false, all: false) >> selectedMetadataMode
        metadataMode = selectedMetadataMode
    }

    if(metadataMode == "Review metadata") {
        matchText(text: "${canonicalStem}", regex: "^(?!(?:NEH[0-9]|Ps[0-9]|OH[0-9]|[0-9]|Hy-|RSC-))(?!.*-Hebdom)[^-]+-.+$", caseSensitive: false) >> composerTitleMatch
        var composerSuggestion = "-"
        var titleSuggestion = "${canonicalStem}"
        if(composerTitleMatch != nil) {
            replaceText(input: "${canonicalStem}", find: "^([^-]+)-(.+)$", replace: "$1", isRegex: true, caseSensitive: true) >> matchedComposerSuggestion
            composerSuggestion = matchedComposerSuggestion
            replaceText(input: "${canonicalStem}", find: "^([^-]+)-(.+)$", replace: "$2", isRegex: true, caseSensitive: true) >> matchedTitleSuggestion
            titleSuggestion = matchedTitleSuggestion
        }
        replaceText(input: "${titleSuggestion}", find: "([a-z])([A-Z])", replace: "$1 $2", isRegex: true, caseSensitive: true) >> spacedTitle
        replaceText(input: "${spacedTitle}", find: "-", replace: " - ", isRegex: false, caseSensitive: true) >> readableTitle
        askForInput(prompt: "Confirm title, retaining movement and variant details.", type: Text, default: "${readableTitle}", allowDecimal: false, allowNegative: false) >> enteredTitle
        replaceText(input: "${enteredTitle}", find: "^[ ]+|[ ]+$", replace: "", isRegex: true, caseSensitive: true) >> displayTitle
        if(displayTitle == nil) {
            text(text: "Empty title. Rerun and choose Review metadata to finish any imported score.") >> stopMessage16
            exit(var: stopMessage16)
        }
        askForInput(prompt: "Confirm composer; a single hyphen means unknown/not applicable.", type: Text, default: "${composerSuggestion}", allowDecimal: false, allowNegative: false) >> enteredComposer
        replaceText(input: "${enteredComposer}", find: "^[ ]+|[ ]+$", replace: "", isRegex: true, caseSensitive: true) >> composerName

        // FS04 RENAME. Replace next Nothing with Rename score. Scores=named native workingScore variable, Name=displayTitle Text. No output from Rename is consumed; workingScore stays native.
        nothing()
        if(composerName != nil) {
            if(composerName != "-") {
                // FS05 COMPOSER LOOKUP. Replace next Nothing with Find composer (expanded Get composer named Name). Name=composerName; Continue if not found ON. Reconnect the following If input to its Composer output, and use that output in FS06.
                nothing() >> existingComposer
                if(existingComposer != nil) {
                    // FS06 ASSIGN EXISTING COMPOSER. Replace next Nothing with Add metadata. Score=workingScore; Composers= FS05 Composer output; Bookmarks empty; other categories unset.
                    nothing()
                } else {
                    // FS07 CREATE AND ASSIGN COMPOSER. Replace next Nothing with Create composer named composerName and add to Scores=workingScore; Bookmarks empty. Do not create twice.
                    nothing()
                }
            }
        }
    }
    // COL03 PREPARED SCORES: Replace next Nothing with native Add to Variable, input=named native workingScore, variable=preparedScores. Append exactly one Score per preparation iteration, including deliberate repeats.
    nothing()
}

list(items: ["Create new", "Replace existing"]) >> setlistChoices
choose(list: setlistChoices, prompt: "Create an unused service name, or replace an existing service?", multiple: false, all: false) >> setlistMode
if(setlistMode == "Create new") {
    // FS08 CREATE SETLIST. Replace next Nothing with new setlist (expanded Create Setlist Name). Name=serviceName; Folder unset. Use an unused name. Creation's return is not consumed.
    nothing()
}
// FS09 RETRIEVE SETLISTS. Replace next Nothing with Get all setlists. Reconnect the following Choose from List input to its Setlists output.
nothing() >> availableSetlists
choose(list: availableSetlists, prompt: "Select the new service, or the existing service to replace.", multiple: false, all: false) >> targetSetlist
alert(alert: """
${runningOrder}

Apply this running order to the setlist you selected? Replace existing clears its entries first; scores and clean PDFs remain.
""", title: "Apply running order", cancel: true)
if(setlistMode == "Replace existing") {
    // FS10 CLEAR ONLY THE SELECTED SETLIST. Replace next Nothing with Remove all setlist entries. Setlist=targetSetlist native Chosen Item. This is inside Replace existing and after all scores are prepared.
    nothing()
}
repeatEach(preparedScores) {
    // FS11 ADD ONE SCORE. Replace next Nothing with add scores to setlist. Scores=Repeat Item from THIS writing loop; Setlist=targetSetlist; Skip if already added OFF to retain deliberate repetitions. Loop results from this action are not consumed.
    nothing()
}
// FS12 READBACK. Replace next Nothing with Get setlist entries. Setlist=targetSetlist; Sorting=Manual. Reconnect Count below to its native Entries output; inspect actual Manual order in forScore after FS13.
nothing() >> actualEntries
count(type: Items, input: actualEntries) >> actualCount
count(type: Items, input: preparedScores) >> requestedCount
text(text: """
${serviceName}
${runningOrder}
Service type: ${selectedServiceType}
Requested entries: ${requestedCount}
Readback entries: ${actualCount}
Running order submitted to selected setlist. Check Manual order and metadata; count equality alone is not identity/order verification.
""") >> serviceReport
// FS13 OPEN SERVICE. Replace next Nothing with open setlist. Setlist=targetSetlist. Do not supply serviceName Text instead of the object. Keep the following Stop and Output connected to serviceReport.
nothing()
exit(var: serviceReport)
