// Installed name: UTIL - Canonical Choir Name
// Purpose: Remove a terminal PDF extension and recognised service prefix.
// Role: Reusable; specification revision 5; interface 2.0.
// Input: Native List of exactly two Text items: filename, config.prefixRegex.
// Output: One Text stem, or empty output after invalid input. Caller rejects empty.
// Side effects: None. Direct and called execution have no prompts.
// Dependencies: None.
// Target: iPadOS 26.6; Jellycuts 2026.6.4 reported, bundled compiler unknown.
// Profile: Existing Open-Jellycore candidate; replaceText from official docs.
// Stage: Drafted, not compiled/imported/device tested.
// Contract/tests/install: ../../docs/config-integration.md; ../../tests/config-integration.md

import Shortcuts

count(type: Items, input: ShortcutInput) >> argumentCount
text(text: "${argumentCount}") >> argumentCountText
if(argumentCountText != "2") {
    alert(alert: "Supply a List containing filename Text and config.prefixRegex Text.", title: "Naming input needs updating", cancel: false)
    output(result: "", noResultBehavior: Respond, response: "")
}
getItemFromList(list: ShortcutInput, type: Item at Index, index: "1", startIndex: "", endIndex: "") >> filename
getItemFromList(list: ShortcutInput, type: Item at Index, index: "2", startIndex: "", endIndex: "") >> prefixRegex
if(filename == nil) {
    output(result: "", noResultBehavior: Respond, response: "")
}
if(prefixRegex == nil) {
    alert(alert: "Set a nonempty prefixRegex in config.json.", title: "Missing prefix pattern", cancel: false)
    output(result: "", noResultBehavior: Respond, response: "")
}
replaceText(input: "${filename}", find: "[.]pdf$", replace: "", isRegex: true, caseSensitive: false) >> filenameStem
replaceText(input: "${filenameStem}", find: "${prefixRegex}", replace: "", isRegex: true, caseSensitive: false) >> canonicalName
output(result: "${canonicalName}", noResultBehavior: Respond, response: "${canonicalName}")
