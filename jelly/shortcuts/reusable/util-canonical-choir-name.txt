// RECEIVING CHECK: Preserve the caller's two separate Text items. The incoming-List extraction probe now works by user report. If this helper receives collapsed/missing input, inspect its receiving settings against that working probe; the earlier installed copy used a manual Text receiving setting. Do not assume the full helper's receiving configuration has been verified by the probe.
// MANUAL OUTPUT REPAIR: The final Stop and Output must return the canonicalName Text token from the second Replace Text action, not Shortcut Input, filenameStem or the literal word canonicalName. Positional Output follows the confirmed app form; inspect/reselect the interpolated token after export. This does not establish arbitrary native File returns.

count(type: Items, input: ShortcutInput) >> argumentCount
text(text: "${argumentCount}") >> argumentCountText
if(argumentCountText != "2") {
alert(alert: "Supply a List containing filename Text and config.prefixRegex Text.", title: "Naming input needs updating", cancel: false)
output("")
}
// SOURCE EXTRACTION: Quoted index operands are required by the tested app; unquoted digits fail scope checking. Keep downstream consumers connected to the current named variables, not their empty initial Text actions.
var filename = ""
var prefixRegex = ""
repeatEach(ShortcutInput) {
if(RepeatIndex == "1") {
filename = RepeatItem
}
if(RepeatIndex == "2") {
prefixRegex = RepeatItem
}
}
if(filename == nil) {
output("")
}
if(prefixRegex == nil) {
alert(alert: "Set a nonempty prefixRegex in config.json.", title: "Missing prefix pattern", cancel: false)
output("")
}
text(text: "${filename}") >> filenameText
replaceText(input: "${filenameText}", find: "[.]pdf$", replace: "", isRegex: true, caseSensitive: false) >> filenameStem
replaceText(input: "${filenameStem}", find: "${prefixRegex}", replace: "", isRegex: true, caseSensitive: false) >> canonicalName
output("${canonicalName}")
