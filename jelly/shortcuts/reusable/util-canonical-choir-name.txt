nothing() >> noNameResult

// NAME01 FILENAME: replace next Nothing with native Get Item from List, List=Shortcut Input, Item at Index=1. Reconnect filename in its presence If and the filename Text action input to that native result before deleting Nothing.
nothing() >> filename
// NAME02 PREFIX PATTERN: replace next Nothing with native Get Item from List, List=Shortcut Input, Item at Index=2. Reconnect prefixRegex in its presence If and the second Replace Text Find field to that native result before deleting Nothing.

    alert(alert: "Set a nonempty prefixRegex in config.json.", title: "Missing prefix pattern", cancel: false)
    exit(var: noNameResult)
