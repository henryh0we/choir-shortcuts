// Installed name: UTIL - Archive Choir PDF
// Purpose: Generated archive helper; complete CFG slots after importing.
// Role: Reusable; specification revision 6; completed interface 1.0.
// Input: One native PDF File. Missing/multiple/non-PDF inputs stop without saving.
// Output after assembly: One native File, or empty output after validation alerts.
// Direct/called: May ask for a type folder when absent; no unattended mode.
// Side effects after assembly: Save a canonical clean PDF; never overwrite.
// Dependency: UTIL - Canonical Choir Name interface 1.0, util-canonical-choir-name.jelly.
// Target: iPadOS 26.6, Jellycuts 2026.6.4; selected iCloud archive folder permission.
// Stage: Drafted; local docs/source reviewed, not compiled/device tested.
// Assembly and tests: ../../docs/choir-service-assembly.md; ../../tests/choir-service.md

import Shortcuts

// COMPATIBILITY: After export, set all seven empty-result Stop and Output
// actions to Do Nothing when there is no recipient. Respond is a temporary
// export workaround for the app rejecting the DoNothing enum identifier.

/* ASSEMBLY GUARD: Keep this first Stop and Output until CFG01-CFG04
   are complete. Then remove only the Stop and Output immediately below. */
output(result: "Archive helper needs its CFG assembly slots completed.", noResultBehavior: Respond, response: "Complete archive assembly first.")

if(ShortcutInput == nil) {
    alert(alert: "Supply one PDF File.", title: "No source PDF", cancel: false)
    output(result: "", noResultBehavior: Respond, response: "")
}
count(type: Items, input: ShortcutInput) >> inputCount
text(text: "${inputCount}") >> inputCountText
if(inputCountText != "1") {
    alert(alert: "Supply exactly one PDF File.", title: "One PDF required", cancel: false)
    output(result: "", noResultBehavior: Respond, response: "")
}
fileDetail(input: ShortcutInput, property: File Extension) >> sourceExtension
changeCase(text: "${sourceExtension}", case: lowercase) >> lowerExtension
if(lowerExtension != "pdf") {
    alert(alert: "No file saved. Select a PDF.", title: "PDF required", cancel: false)
    output(result: "", noResultBehavior: Respond, response: "")
}
fileDetail(input: ShortcutInput, property: Name) >> sourceName
runShortcut(name: "UTIL - Canonical Choir Name", input: sourceName, show: false) >> canonicalStem
if(canonicalStem == nil) {
    alert(alert: "A file named only .pdf is not supported.", title: "Empty music identity", cancel: false)
    output(result: "", noResultBehavior: Respond, response: "")
}

/* CFG01 ARCHIVE ROOT. Replace the next Nothing with a native Folder
   action. Select the existing iCloud Choir Music/Library folder once.
   Reconnect BOTH Get Contents of Folder actions to that native Folder
   output (source variable archiveRoot). Do not use a Text path. */
nothing() >> archiveRoot

getFolderContents(folder: archiveRoot, recursive: true) >> archiveContents
// A named variable survives the loop without capturing Repeat Results.
// Keep the native File when exactly one canonical match exists.
// UNVERIFIED: Check both If inputs and CFG03 use the named variable after export.
nothing() >> noArchiveMatch
var existingCleanFile = noArchiveMatch
repeatEach(archiveContents) {
    fileDetail(input: RepeatItem, property: File Extension) >> archiveExtension
    changeCase(text: "${archiveExtension}", case: lowercase) >> archiveExtensionLower
    if(archiveExtensionLower == "pdf") {
        fileDetail(input: RepeatItem, property: Name) >> archiveName
        runShortcut(name: "UTIL - Canonical Choir Name", input: archiveName, show: false) >> archiveStem
        if(archiveStem == "${canonicalStem}") {
            if(existingCleanFile != nil) {
                alert(alert: "Resolve duplicate canonical PDFs in iCloud before retrying.", title: "Multiple clean copies", cancel: false)
                output(result: "", noResultBehavior: Respond, response: "")
            }
            existingCleanFile = RepeatItem
        }
    }
}

if(existingCleanFile != nil) {
    /* CFG03 EXISTING FILE RETURN. In the next Stop and Output,
       replace its EMPTY result field with the named existingCleanFile
       variable (Set Variable in the search loop). No Text conversion. */
    output(result: "", noResultBehavior: Respond, response: "")
}

getFolderContents(folder: archiveRoot, recursive: false) >> typeFolders
choose(list: typeFolders, prompt: "Choose an existing music-type folder for this PDF.", multiple: false, all: false) >> typeFolder
setName(input: ShortcutInput, name: "${canonicalStem}.pdf", dontIncludeExtension: false) >> renamedFile
/* CFG02 SAVE DESTINATION. The next Save File is generated with its picker
   enabled because the reviewed binding cannot encode a Folder destination.
   Set Ask Where to Save OFF and Destination to native typeFolder from
   Choose from List above. Keep Overwrite OFF and the native renamedFile input. */
saveFile(input: renamedFile, ask: true, path: "", overwrite: false) >> savedCleanFile
/* CFG04 NEW FILE RETURN. In the next Stop and Output, replace its EMPTY
   result field with native savedCleanFile from Save File above.
   Do not interpolate it into Text or use its filename property. */
output(result: "", noResultBehavior: Respond, response: "")
